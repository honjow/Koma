cmake_minimum_required(VERSION 3.14)
project(KomaImageProcessorExecution C CXX ASM)
get_filename_component(KOMA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(WAMR_ROOT_DIR "${KOMA_ROOT}/third_party/wasm-micro-runtime")
string(TOLOWER "${CMAKE_HOST_SYSTEM_NAME}" WAMR_BUILD_PLATFORM)
set(WAMR_BUILD_TARGET "AARCH64")
if(NOT CMAKE_HOST_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
    set(WAMR_BUILD_TARGET "X86_64")
endif()
set(WAMR_BUILD_INTERP 1)
set(WAMR_BUILD_AOT 0)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_FAST_JIT 0)
set(WAMR_BUILD_LIBC_BUILTIN 1)
set(WAMR_BUILD_LIBC_WASI 0)
set(WAMR_BUILD_MULTI_MODULE 0)
set(WAMR_BUILD_REF_TYPES 1)
set(WAMR_BUILD_BULK_MEMORY 1)
set(WAMR_BUILD_THREAD_MGR 1)
set(WAMR_BUILD_LIB_PTHREAD 0)
set(WAMR_DISABLE_HW_BOUND_CHECK 1)
include("${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake")
add_library(test_wamr STATIC ${WAMR_RUNTIME_LIB_SOURCE})
target_compile_definitions(test_wamr PUBLIC WASM_ENABLE_HEAP_AUX_STACK_ALLOCATION=1)
target_include_directories(test_wamr PUBLIC "${WAMR_ROOT_DIR}/core/iwasm/include")
add_executable(image_processor_runner runner.cpp "${KOMA_ROOT}/entry/src/main/cpp/image_processor_host.cpp")
target_compile_features(image_processor_runner PRIVATE cxx_std_17)
target_compile_definitions(image_processor_runner PRIVATE KOMA_ENABLE_WAMR=1)
target_include_directories(image_processor_runner PRIVATE "${KOMA_ROOT}/entry/src/main/cpp")
target_link_libraries(image_processor_runner PRIVATE test_wamr pthread m)
